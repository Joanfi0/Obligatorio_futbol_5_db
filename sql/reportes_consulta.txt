-- Consultas de referencia: sustituir %s por el ID de edición (ej. 1).

-- Tabla de posiciones
SELECT nombre,jugados,ganados,empatados,perdidos,goles_favor,goles_contra,diferencia,puntos FROM v_posiciones WHERE id_torneo=%s ORDER BY puntos DESC,diferencia DESC,goles_favor DESC,nombre;

-- Diez goleadores
SELECT j.ci,j.nombre,j.apellido,e.nombre equipo,COUNT(*) goles FROM gol g JOIN partido p ON p.id_partido=g.id_partido JOIN jugador j ON j.ci=g.ci_jugador JOIN planilla_partido pp ON pp.id_partido=g.id_partido AND pp.ci_jugador=g.ci_jugador JOIN equipo e ON e.id_equipo=pp.id_equipo WHERE p.id_torneo=%s AND p.estado='jugado' AND g.autogol=0 GROUP BY j.ci,j.nombre,j.apellido,e.nombre ORDER BY goles DESC,j.ci LIMIT 10;

-- Fair play
SELECT e.nombre,COALESCE(x.amarillas,0) amarillas,COALESCE(x.rojas,0) rojas,COALESCE(x.puntos,0) puntos FROM equipo e LEFT JOIN (SELECT pp.id_equipo,SUM(t.color='amarilla') amarillas,SUM(t.color='roja') rojas,SUM(IF(t.color='amarilla',1,3)) puntos FROM tarjeta t JOIN partido p ON p.id_partido=t.id_partido JOIN planilla_partido pp ON pp.id_partido=t.id_partido AND pp.ci_jugador=t.ci_jugador WHERE p.estado='jugado' GROUP BY pp.id_equipo) x ON x.id_equipo=e.id_equipo WHERE e.id_torneo=%s ORDER BY puntos,e.nombre;

-- Promedio de goles por fecha y edición
SELECT IF(GROUPING(numero_fecha),'TOTAL',CAST(numero_fecha AS CHAR)) fecha,COUNT(*) partidos,ROUND(AVG(goles_local+goles_visitante),2) promedio FROM v_resultados WHERE id_torneo=%s GROUP BY numero_fecha WITH ROLLUP;

-- Uso de canchas y turnos
SELECT c.nombre complejo,p.nombre_cancha,t.hora_inicio,COUNT(*) partidos_jugados FROM partido p JOIN complejo c ON c.id_complejo=p.id_complejo JOIN turno t ON t.id_turno=p.id_turno WHERE p.id_torneo=%s AND p.estado='jugado' GROUP BY c.id_complejo,c.nombre,p.nombre_cancha,t.hora_inicio ORDER BY partidos_jugados DESC,complejo,p.nombre_cancha,t.hora_inicio;

-- Jugadores inscriptos por facultad y tipo
SELECT f.nombre facultad,j.tipo,COUNT(*) cantidad FROM jugador j JOIN facultad f ON f.id_facultad=j.id_facultad JOIN equipo_jugador_torneo ej ON ej.ci_jugador=j.ci WHERE ej.id_torneo=%s GROUP BY f.id_facultad,f.nombre,j.tipo ORDER BY facultad,tipo;

-- Inscriptos que nunca figuraron en una planilla
SELECT j.ci,j.nombre,j.apellido,e.nombre equipo FROM equipo_jugador_torneo ej JOIN jugador j ON j.ci=ej.ci_jugador JOIN equipo e ON e.id_equipo=ej.id_equipo WHERE ej.id_torneo=%s AND NOT EXISTS(SELECT 1 FROM planilla_partido pp JOIN partido p ON p.id_partido=pp.id_partido WHERE pp.ci_jugador=ej.ci_jugador AND p.id_torneo=ej.id_torneo) ORDER BY equipo,j.ci;

-- Sanciones por equipo y motivo
SELECT e.nombre equipo,s.motivo,COUNT(*) cantidad FROM sancion_jugador s JOIN equipo_jugador_torneo ej ON ej.ci_jugador=s.ci_jugador AND ej.id_torneo=s.id_torneo JOIN equipo e ON e.id_equipo=ej.id_equipo WHERE s.id_torneo=%s GROUP BY e.id_equipo,e.nombre,s.motivo ORDER BY equipo,s.motivo;

-- Porcentaje de partidos por estado
SELECT estado,COUNT(*) cantidad,ROUND(100.0*COUNT(*)/SUM(COUNT(*)) OVER(),2) porcentaje FROM partido WHERE id_torneo=%s GROUP BY estado ORDER BY estado;

-- Extra: equipos con menos goles recibidos
SELECT nombre,jugados,goles_contra,ROUND(goles_contra/NULLIF(jugados,0),2) promedio_recibido FROM v_posiciones WHERE id_torneo=%s AND jugados>0 ORDER BY promedio_recibido,nombre;

-- Extra: suspensiones por fecha
SELECT s.ci_jugador,j.nombre,j.apellido,s.motivo,s.numero_fecha_inicio,s.numero_fecha_inicio+s.cantidad_fechas-1 fecha_fin FROM sancion_jugador s JOIN jugador j ON j.ci=s.ci_jugador WHERE s.id_torneo=%s ORDER BY numero_fecha_inicio,j.ci;

-- Extra: composición de planteles
SELECT e.nombre,COUNT(ej.ci_jugador) jugadores,COALESCE(SUM(j.tipo IN ('docente','funcionario')),0) docentes_funcionarios FROM equipo e LEFT JOIN equipo_jugador_torneo ej ON ej.id_equipo=e.id_equipo AND ej.id_torneo=e.id_torneo LEFT JOIN jugador j ON j.ci=ej.ci_jugador WHERE e.id_torneo=%s GROUP BY e.id_equipo,e.nombre ORDER BY e.nombre;
